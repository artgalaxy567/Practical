import heapq


import heapq

class Node:
    def __init__(self, freq, char=None, left=None, right=None):
        self.freq = freq
        self.char = char
        self.left = left
        self.right = right

    def __lt__(self, other):
        return self.freq < other.freq


def generate_codes(node, code="", huffman_code=None):
    if huffman_code is None:
        huffman_code = {}

    if node is None:
        return huffman_code

    if node.char is not None:
        huffman_code[node.char] = code

    generate_codes(node.left, code + "0", huffman_code)
    generate_codes(node.right, code + "1", huffman_code)

    return huffman_code


chars = {
    'A': 7,
    'B': 15,
    'C': 20,
    'D': 13,
    'E': 25,
    'F': 45
}

heap = []

for char, freq in chars.items():
    heapq.heappush(heap, Node(freq, char))

while len(heap) > 1:
    left = heapq.heappop(heap)
    right = heapq.heappop(heap)

    merged = Node(left.freq + right.freq,
                  left=left, right=right)

    heapq.heappush(heap, merged)

root = heap[0]
codes = generate_codes(root)

print("Huffman Codes")
print("---------------------------")

for char in sorted(codes):
    print(char, ":", codes[char])

total_frequency = sum(chars.values())
total_bits = 0

print("\nEncoded Length of Each Character")
print("----------------------------------")

for char in sorted(chars):
    code_length = len(codes[char])
    bits = chars[char] * code_length
    total_bits += bits

    print(char,
          "Frequency =", chars[char],
          "Code =", codes[char],
          "Bits =", bits)

average_code_length = total_bits / total_frequency

print("\nTotal Frequency =", total_frequency)
print("Average Code Length =", round(average_code_length, 2), "bits")
print("Total Huffman Encoded Message Length =", total_bits, "bits")

def fractional_knapsack(profit, weight, capacity):
    n = len(profit)
    items = []
    for i in range(n):
        ratio = profit[i] / weight[i]
        items.append((ratio, profit[i], weight[i]))
    items.sort(reverse=True)
    total_profit = 0.0
    for ratio, p, w in items:
        if capacity >= w:
            total_profit += p
            capacity -= w
        else:
            total_profit += ratio * capacity
            break
    return total_profit
n = int(input("Enter number of items: "))

        f"Enter profit and weight of item {i + 1}: "

    weight.append(w)

Write a program to solve a 0-1 Knapsack problem using dynamic programming or branch and bound strategy.

def knapSack(W, wt, val, n):
    dp = [0 for i in range(W + 1)]
    for i in range(1, n + 1):
        for w in range(W, 0, -1):
            if wt[i - 1] <= w:
                dp[w] = max(
                    dp[w],
                    dp[w - wt[i - 1]] + val[i - 1]
                )

    return dp[W]
n = int(input("Enter number of items: "))
val = list(map(int, input("Enter profits: ").split()))
wt = list(map(int, input("Enter weights: ").split()))

elif W < 0 or any(w <= 0 for w in wt):
    print("Error: Capacity must be non-negative and weights must be positive.")
else:
    print("Maximum profit =", knapSack(W, wt, val, n))

class NQueens:
    def __init__(self) -> None:
        self.size = int(input("Enter size of chessboard: "))
        self.board = [[False]*self.size for _ in range(self.size)]
        self.count = 0
    def printBoard(self):
        for row in self.board:
            for ele in row:
                if ele == True:
                    print("Q",end=" ")
                else:
                    print("X",end=" ")
            print()
        print()
    def isSafe(self,row:int,col:int) -> bool:
        for i in self.board:
            if i[col] == True:
                return False
        i = row
        j = col
        while i >= 0 and j >= 0:
            if self.board[i][j] == True:
                return False
            i -= 1
            j -= 1
        i = row
        j = col
        while i < self.size and j < self.size:
            if self.board[i][j] == True:
                return False
            i += 1
            j += 1
        i = row
        j = col
        while i >= 0 and j < self.size:
            if self.board[i][j] == True:
                return False
            i -= 1
            j += 1
        i = row
        j = col
        while i < self.size and j >= 0:
            if self.board[i][j] == True:
                return False
            i += 1
            j -= 1
        return True
    def set_position_first_queen(self):
        print("Enter coordinates of first queen: ")
        row = int(input(f"Enter row (1-{self.size}): "))
        col = int(input(f"Enter column (1-{self.size}): "))
        self.board[row-1][col-1] = True
        self.printBoard()
    def solve(self,row:int):
        if row == self.size:
            self.count += 1
            self.printBoard()
            return
        if any(self.board[row]) is True:
            self.solve(row+1)
            return

        for col in range(self.size):
            if self.isSafe(row,col) == True:
                self.board[row][col] = True
                self.solve(row+1)
                self.board[row][col] = False
    def displayMessage(self):
        if self.count > 0:
            print("Solution exists for the given position of the queen.")
        else:
            print("Solution doesn't exist for the given position of the queen.")
